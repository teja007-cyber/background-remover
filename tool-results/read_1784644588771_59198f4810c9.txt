    84→function formatFileSize(bytes: number) {
    85→  if (bytes < 1024) return `${bytes} B`
    86→  if (bytes < 1024 * 1024) return `${(bytes / 1024).toFixed(1)} KB`
    87→  return `${(bytes / (1024 * 1024)).toFixed(1)} MB`
    88→}

    90→function generateId() {
    91→  return Math.random().toString(36).slice(2, 10)

    94→function readFileAsDataUrl(file: File): Promise<string> {
    95→  return new Promise((resolve, reject) => {
    96→    const reader = new FileReader()
    97→    reader.onload = (e) => resolve(e.target?.result as string)
    98→    reader.onerror = reject
    99→    reader.readAsDataURL(file)
   100→  })
   101→}
