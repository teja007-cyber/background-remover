    94→    94→function readFileAsDataUrl(file: File): Promise<string> {
    95→    95→  return new Promise((resolve, reject) => {
    96→    96→    const reader = new FileReader()
    97→    97→    reader.onload = (e) => resolve(e.target?.result as string)
    98→    98→    reader.onerror = reject
    99→    99→    reader.readAsDataURL(file)
   100→   100→  })
   101→   101→}
