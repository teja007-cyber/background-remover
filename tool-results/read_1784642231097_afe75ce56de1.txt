     1→'use client'
     2→
     3→import { useState, useRef, useCallback, useEffect } from 'react'
     4→import { motion, AnimatePresence } from 'framer-motion'
     5→import {
     6→  Eraser,
     7→  Upload,
     8→  Download,
     9→  RefreshCw,
    10→  ImageIcon,
    11→  Loader2,
    12→  Sparkles,
    13→  CheckCircle2,
    14→  AlertTriangle,
    15→  ShieldCheck,
    16→  Zap,
    17→  Eye,
    18→  Scissors,
    19→} from 'lucide-react'
    20→import { toast, Toaster } from 'sonner'
    21→import {
    22→  Card,
    23→  CardContent,
    24→  CardDescription,
    25→  CardHeader,
    26→  CardTitle,
    27→} from '@/components/ui/card'
    28→import { Button } from '@/components/ui/button'
    29→import { Badge } from '@/components/ui/badge'
    30→import { Progress } from '@/components/ui/progress'
    31→import { Separator } from '@/components/ui/separator'
    32→
    33→type AppState = 'idle' | 'processing' | 'done' | 'error'
    34→
    35→const CHECKERBOARD_BG = {
    36→  backgroundImage: [
    37→    'linear-gradient(45deg, #e0e0e0 25%, transparent 25%)',
    38→    'linear-gradient(-45deg, #e0e0e0 25%, transparent 25%)',
    39→    'linear-gradient(45deg, transparent 75%, #e0e0e0 75%)',
    40→    'linear-gradient(-45deg, transparent 75%, #e0e0e0 75%)',
    41→  ].join(', '),
    42→  backgroundSize: '20px 20px',
    43→  backgroundPosition: '0 0, 0 10px, 10px -10px, -10px 0px',
    44→  backgroundColor: '#ffffff',
    45→}
    46→
    47→const VALID_TYPES = ['image/png', 'image/jpeg', 'image/webp']
    48→const MAX_FILE_SIZE = 10 * 1024 * 1024 // 10MB
    49→
    50→export default function Home() {
    51→  const [state, setState] = useState<AppState>('idle')
    52→  const [originalImage, setOriginalImage] = useState<string | null>(null)
    53→  const [processedImage, setProcessedImage] = useState<string | null>(null)
    54→  const [progress, setProgress] = useState(0)
    55→  const [fileName, setFileName] = useState('')
    56→  const [fileSize, setFileSize] = useState(0)
    57→  const [imageDimensions, setImageDimensions] = useState<{ w: number; h: number } | null>(null)
    58→  const [isDragging, setIsDragging] = useState(false)
    59→  const [errorMsg, setErrorMsg] = useState('')
    60→  const [processingTime, setProcessingTime] = useState(0)
    61→  const fileInputRef = useRef<HTMLInputElement>(null)
    62→  const abortControllerRef = useRef<AbortController | null>(null)
    63→
    64→  // Cleanup on unmount
    65→  useEffect(() => {
    66→    return () => {
    67→      if (abortControllerRef.current) {
    68→        abortControllerRef.current.abort()
    69→      }
    70→    }
    71→  }, [])
    72→
    73→  const handleFile = useCallback(async (file: File) => {
    74→    // Validate type
    75→    if (!VALID_TYPES.includes(file.type)) {
    76→      toast.error('Invalid file type', {
    77→        description: 'Please upload a PNG, JPG, JPEG, or WebP image.',
    78→      })
    79→      return
    80→    }
    81→
    82→    // Validate size
    83→    if (file.size > MAX_FILE_SIZE) {
    84→      toast.error('File too large', {
    85→        description: 'Please upload an image smaller than 10MB.',
    86→      })
    87→      return
    88→    }
    89→
    90→    // Preview original
    91→    const reader = new FileReader()
    92→    reader.onload = (e) => {
    93→      const dataUrl = e.target?.result as string
    94→      setOriginalImage(dataUrl)
    95→      // Get image dimensions
    96→      const img = new window.Image()
    97→      img.onload = () => {
    98→        setImageDimensions({ w: img.naturalWidth, h: img.naturalHeight })
    99→      }
   100→      img.src = dataUrl
   101→    }
   102→    reader.readAsDataURL(file)
   103→
   104→    setFileName(file.name)
   105→    setFileSize(file.size)
   106→    setState('processing')
   107→    setProgress(0)
   108→    setErrorMsg('')
   109→    setProcessingTime(0)
   110→
   111→    // Simulate progress while waiting for API
   112→    let currentProgress = 0
   113→    const startTime = Date.now()
   114→    const progressInterval = setInterval(() => {
   115→      currentProgress += Math.random() * 6
   116→      if (currentProgress >= 85) {
   117→        currentProgress = 85
   118→        clearInterval(progressInterval)
   119→      }
   120→      setProgress(Math.min(currentProgress, 85))
   121→    }, 500)
   122→
   123→    try {
   124→      const formData = new FormData()
   125→      formData.append('file', file)
   126→
   127→      abortControllerRef.current = new AbortController()
   128→
   129→      const response = await fetch('/api/remove-bg', {
   130→        method: 'POST',
   131→        body: formData,
   132→        signal: abortControllerRef.current.signal,
   133→      })
   134→
   135→      const data = await response.json()
   136→      clearInterval(progressInterval)
   137→
   138→      if (!response.ok) {
   139→        // Server returned an error
   140→        setState('error')
   141→        setErrorMsg(data.error || `Server error (${response.status})`)
   142→        toast.error('Processing failed', {
   143→          description: data.error || 'Something went wrong. Please try again.',
   144→        })
   145→        return
   146→      }
   147→
   148→      if (data.success && data.image) {
   149→        const elapsed = ((Date.now() - startTime) / 1000).toFixed(1)
   150→        setProcessingTime(Number(elapsed))
   151→        setProgress(100)
   152→        setTimeout(() => {
   153→          setProcessedImage(data.image)
   154→          setState('done')
   155→        }, 300)
   156→        toast.success('Background removed successfully!', {
   157→          description: `Processed in ${elapsed}s`,
   158→        })
   159→      } else {
   160→        setState('error')
   161→        setErrorMsg(data.error || 'Unexpected response from server')
   162→        toast.error('Processing failed', {
   163→          description: data.error || 'Something went wrong. Please try again.',
   164→        })
   165→      }
   166→    } catch (err) {
   167→      clearInterval(progressInterval)
   168→      if (err instanceof DOMException && err.name === 'AbortError') return
   169→      setState('error')
   170→      setErrorMsg('Failed to connect to the server. Please check your connection and try again.')
   171→      toast.error('Network error', {
   172→        description: 'Failed to connect to the server.',
   173→      })
   174→    }
   175→  }, [])
   176→
   177→  const handleDrop = useCallback(
   178→    (e: React.DragEvent) => {
   179→      e.preventDefault()
   180→      e.stopPropagation()
   181→      setIsDragging(false)
   182→      const file = e.dataTransfer.files[0]
   183→      if (file) handleFile(file)
   184→    },
   185→    [handleFile]
   186→  )
   187→
   188→  const handleDragOver = useCallback((e: React.DragEvent) => {
   189→    e.preventDefault()
   190→    e.stopPropagation()
   191→    setIsDragging(true)
   192→  }, [])
   193→
   194→  const handleDragLeave = useCallback((e: React.DragEvent) => {
   195→    e.preventDefault()
   196→    e.stopPropagation()
   197→    setIsDragging(false)
   198→  }, [])
   199→
   200→  const handleClick = useCallback(() => {
   201→    fileInputRef.current?.click()
   202→  }, [])
   203→
   204→  const handleInputChange = useCallback(
   205→    (e: React.ChangeEvent<HTMLInputElement>) => {
   206→      const file = e.target.files?.[0]
   207→      if (file) handleFile(file)
   208→    },
   209→    [handleFile]
   210→  )
   211→
   212→  const handleDownload = useCallback(() => {
   213→    if (!processedImage) return
   214→    const link = document.createElement('a')
   215→    link.href = processedImage
   216→    link.download = fileName.replace(/\.[^.]+$/, '') + '-no-bg.png'
   217→    document.body.appendChild(link)
   218→    link.click()
   219→    document.body.removeChild(link)
   220→    toast.success('Download started!', {
   221→      description: 'Your image is being saved as PNG.',
   222→    })
   223→  }, [processedImage, fileName])
   224→
   225→  const handleReset = useCallback(() => {
   226→    if (abortControllerRef.current) {
   227→      abortControllerRef.current.abort()
   228→    }
   229→    setState('idle')
   230→    setOriginalImage(null)
   231→    setProcessedImage(null)
   232→    setProgress(0)
   233→    setFileName('')
   234→    setFileSize(0)
   235→    setImageDimensions(null)
   236→    setErrorMsg('')
   237→    setProcessingTime(0)
   238→    if (fileInputRef.current) fileInputRef.current.value = ''
   239→  }, [])
   240→
   241→  const formatFileSize = (bytes: number) => {
   242→    if (bytes < 1024) return `${bytes} B`
   243→    if (bytes < 1024 * 1024) return `${(bytes / 1024).toFixed(1)} KB`
   244→    return `${(bytes / (1024 * 1024)).toFixed(1)} MB`
   245→  }
   246→
   247→  return (
   248→    <div className="min-h-screen flex flex-col bg-gradient-to-b from-slate-50 to-slate-100 dark:from-slate-950 dark:to-slate-900">
   249→      <Toaster position="top-center" richColors closeButton />
   250→
   251→      {/* Skip link for accessibility */}
   252→      <a
   253→        href="#main-content"
   254→        className="sr-only focus:not-sr-only focus:absolute focus:top-4 focus:left-4 focus:z-[100] focus:px-4 focus:py-2 focus:bg-emerald-600 focus:text-white focus:rounded-lg"
   255→      >
   256→        Skip to main content
   257→      </a>
   258→
   259→      {/* Header */}
   260→      <header role="banner" className="sticky top-0 z-50 w-full border-b bg-white/80 backdrop-blur-md dark:bg-slate-950/80">
   261→        <nav className="container mx-auto flex h-16 items-center justify-between px-4 sm:px-6 lg:px-8">
   262→          <motion.div
   263→            className="flex items-center gap-3"
   264→            initial={{ opacity: 0, x: -20 }}
   265→            animate={{ opacity: 1, x: 0 }}
   266→            transition={{ duration: 0.4 }}
   267→          >
   268→            <div className="flex h-10 w-10 items-center justify-center rounded-xl bg-emerald-600 text-white shadow-md" aria-hidden="true">
   269→              <Eraser className="h-5 w-5" />
   270→            </div>
   271→            <div>
   272→              <h1 className="text-xl font-bold tracking-tight text-slate-900 dark:text-slate-50">
   273→                BG Remover
   274→              </h1>
   275→              <p className="text-xs text-slate-500 dark:text-slate-400 hidden sm:block">
   276→                Remove image backgrounds instantly
   277→              </p>
   278→            </div>
   279→          </motion.div>
   280→          <Badge
   281→            variant="secondary"
   282→            className="hidden sm:flex gap-1.5 px-3 py-1 text-xs"
   283→          >
   284→            <Sparkles className="h-3 w-3 text-emerald-600" aria-hidden="true" />
   285→            AI Powered
   286→          </Badge>
   287→        </nav>
   288→      </header>
   289→
   290→      {/* Main Content */}
   291→      <main id="main-content" role="main" className="flex-1 container mx-auto px-4 sm:px-6 lg:px-8 py-6 sm:py-10">
   292→        <AnimatePresence mode="wait">
   293→          {/* IDLE STATE - Upload Zone */}
   294→          {state === 'idle' && (
   295→            <motion.section
   296→              key="upload"
   297→              aria-label="Image upload"
   298→              initial={{ opacity: 0, y: 20 }}
   299→              animate={{ opacity: 1, y: 0 }}
   300→              exit={{ opacity: 0, y: -20 }}
   301→              transition={{ duration: 0.3 }}
   302→              className="mx-auto max-w-2xl"
   303→            >
   304→              <div className="text-center mb-8">
   305→                <motion.h2
   306→                  className="text-2xl sm:text-3xl lg:text-4xl font-bold text-slate-900 dark:text-slate-50 mb-3"
   307→                  initial={{ opacity: 0 }}
   308→                  animate={{ opacity: 1 }}
   309→                  transition={{ delay: 0.1 }}
   310→                >
   311→                  Remove Any Background
   312→                </motion.h2>
   313→                <motion.p
   314→                  className="text-slate-600 dark:text-slate-400 text-base sm:text-lg max-w-lg mx-auto"
   315→                  initial={{ opacity: 0 }}
   316→                  animate={{ opacity: 1 }}
   317→                  transition={{ delay: 0.2 }}
   318→                >
   319→                  Upload an image and let AI remove the background in seconds.
   320→                  Full subject preserved — nothing gets cut off.
   321→                </motion.p>
   322→              </div>
   323→
   324→              <Card className="overflow-hidden shadow-lg border-slate-200 dark:border-slate-800">
   325→                <CardContent className="p-0">
   326→                  <div
   327→                    onClick={handleClick}
   328→                    onDrop={handleDrop}
   329→                    onDragOver={handleDragOver}
   330→                    onDragLeave={handleDragLeave}
   331→                    role="button"
   332→                    tabIndex={0}
   333→                    aria-label="Upload image: click or drag and drop"
   334→                    onKeyDown={(e) => {
   335→                      if (e.key === 'Enter' || e.key === ' ') {
   336→                        e.preventDefault()
   337→                        handleClick()
   338→                      }
   339→                    }}
   340→                    className={`
   341→                      relative cursor-pointer transition-all duration-300 ease-in-out
   342→                      border-2 border-dashed rounded-lg m-6 p-8 sm:p-12
   343→                      flex flex-col items-center justify-center gap-4
   344→                      min-h-[280px] sm:min-h-[320px]
   345→                      focus-visible:outline-2 focus-visible:outline-emerald-500 focus-visible:outline-offset-2
   346→                      ${
   347→                        isDragging
   348→                          ? 'border-emerald-500 bg-emerald-50 dark:bg-emerald-950/30 scale-[1.02]'
   349→                          : 'border-slate-300 dark:border-slate-700 hover:border-emerald-400 hover:bg-slate-50 dark:hover:bg-slate-900/50'
   350→                      }
   351→                    `}
   352→                  >
   353→                    <motion.div
   354→                      className={`
   355→                        flex h-16 w-16 items-center justify-center rounded-2xl transition-colors duration-300
   356→                        ${
   357→                          isDragging
   358→                            ? 'bg-emerald-600 text-white'
   359→                            : 'bg-slate-100 text-slate-400 dark:bg-slate-800 dark:text-slate-500'
   360→                        }
   361→                      `}
   362→                      aria-hidden="true"
   363→                      animate={isDragging ? { scale: 1.1, rotate: 5 } : { scale: 1, rotate: 0 }}
   364→                      transition={{ type: 'spring', stiffness: 300, damping: 20 }}
   365→                    >
   366→                      <Upload className="h-7 w-7" />
   367→                    </motion.div>
   368→
   369→                    <div className="text-center space-y-1.5">
   370→                      <p className="text-lg font-semibold text-slate-700 dark:text-slate-200">
   371→                        {isDragging ? 'Drop your image here' : 'Drop your image here or click to browse'}
   372→                      </p>
   373→                      <p className="text-sm text-slate-500 dark:text-slate-400">
   374→                        Select an image file from your device
   375→                      </p>
   376→                    </div>
   377→
   378→                    <div className="flex flex-wrap items-center justify-center gap-1.5 mt-2">
   379→                      {['PNG', 'JPG', 'JPEG', 'WebP'].map((fmt) => (
   380→                        <Badge
   381→                          key={fmt}
   382→                          variant="outline"
   383→                          className="text-xs font-mono text-slate-500 dark:text-slate-400 border-slate-200 dark:border-slate-700"
   384→                        >
   385→                          {fmt}
   386→                        </Badge>
   387→                      ))}
   388→                    </div>
   389→
   390→                    <p className="text-xs text-slate-400 dark:text-slate-500 mt-1">
   391→                      Max file size: 10MB
   392→                    </p>
   393→
   394→                    <input
   395→                      ref={fileInputRef}
   396→                      type="file"
   397→                      accept=".png,.jpg,.jpeg,.webp"
   398→                      onChange={handleInputChange}
   399→                      className="hidden"
   400→                      aria-label="Upload image file"
   401→                    />
   402→                  </div>
   403→                </CardContent>
   404→              </Card>
   405→
   406→              {/* Feature badges */}
   407→              <div className="mt-8 grid grid-cols-2 sm:grid-cols-4 gap-3">
   408→                {[
   409→                  { icon: ShieldCheck, label: '100% Free' },
   410→                  { icon: Zap, label: 'No Signup' },
   411→                  { icon: Eye, label: 'Full Subject Preserved' },
   412→                  { icon: Scissors, label: 'Clean Edges' },
   413→                ].map(({ icon: Icon, label }) => (
   414→                  <div
   415→                    key={label}
   416→                    className="flex items-center justify-center gap-1.5 text-xs text-slate-500 dark:text-slate-400 bg-white dark:bg-slate-900 rounded-lg py-2 px-3 border border-slate-100 dark:border-slate-800"
   417→                  >
   418→                    <Icon className="h-3.5 w-3.5 text-emerald-500" aria-hidden="true" />
   419→                    <span>{label}</span>
   420→                  </div>
   421→                ))}
   422→              </div>
   423→
   424→              {/* How it works */}
   425→              <section aria-label="How it works" className="mt-10">
   426→                <h3 className="text-center text-sm font-semibold text-slate-900 dark:text-slate-50 mb-4">
   427→                  How It Works
   428→                </h3>
   429→                <div className="grid grid-cols-1 sm:grid-cols-3 gap-4">
   430→                  {[
   431→                    { step: '1', title: 'Upload', desc: 'Drag & drop or click to select any image' },
   432→                    { step: '2', title: 'Process', desc: 'AI removes the background while preserving your subject' },
   433→                    { step: '3', title: 'Download', desc: 'Get a transparent PNG ready for any use' },
   434→                  ].map(({ step, title, desc }) => (
   435→                    <div key={step} className="text-center p-4 rounded-lg bg-white dark:bg-slate-900 border border-slate-100 dark:border-slate-800">
   436→                      <div className="flex h-8 w-8 items-center justify-center rounded-full bg-emerald-100 dark:bg-emerald-900/30 text-emerald-700 dark:text-emerald-300 text-sm font-bold mx-auto mb-2">
   437→                        {step}
   438→                      </div>
   439→                      <p className="text-sm font-medium text-slate-900 dark:text-slate-50">{title}</p>
   440→                      <p className="text-xs text-slate-500 dark:text-slate-400 mt-1">{desc}</p>
   441→                    </div>
   442→                  ))}
   443→                </div>
   444→              </section>
   445→            </motion.section>
   446→          )}
   447→
   448→          {/* PROCESSING STATE */}
   449→          {state === 'processing' && (
   450→            <motion.section
   451→              key="processing"
   452→              aria-label="Processing image"
   453→              aria-live="polite"
   454→              initial={{ opacity: 0, y: 20 }}
   455→              animate={{ opacity: 1, y: 0 }}
   456→              exit={{ opacity: 0, y: -20 }}
   457→              transition={{ duration: 0.3 }}
   458→              className="mx-auto max-w-2xl"
   459→            >
   460→              <Card className="overflow-hidden shadow-lg border-slate-200 dark:border-slate-800">
   461→                <CardHeader className="pb-2">
   462→                  <CardTitle className="text-lg text-slate-900 dark:text-slate-50 flex items-center gap-2">
   463→                    <Loader2 className="h-5 w-5 text-emerald-600 animate-spin" aria-hidden="true" />
   464→                    Processing Your Image
   465→                  </CardTitle>
   466→                  <CardDescription>
   467→                    AI is removing the background from &ldquo;{fileName}&rdquo; ({formatFileSize(fileSize)})
   468→                    {imageDimensions && ` — ${imageDimensions.w}×${imageDimensions.h}`}
   469→                  </CardDescription>
   470→                </CardHeader>
   471→                <CardContent className="space-y-6">
   472→                  {/* Original image preview */}
   473→                  {originalImage && (
   474→                    <div className="relative overflow-hidden rounded-lg border border-slate-200 dark:border-slate-700 bg-slate-50 dark:bg-slate-900">
   475→                      <img
   476→                        src={originalImage}
   477→                        alt="Original image being processed"
   478→                        className="w-full h-auto max-h-[400px] object-contain mx-auto opacity-60"
   479→                        width={imageDimensions?.w}
   480→                        height={imageDimensions?.h}
   481→                      />
   482→                      {/* Overlay */}
   483→                      <div className="absolute inset-0 flex items-center justify-center bg-slate-900/20 dark:bg-slate-900/40">
   484→                        <div className="flex flex-col items-center gap-3 bg-white/90 dark:bg-slate-900/90 rounded-2xl px-6 py-5 shadow-xl backdrop-blur-sm">
   485→                          <Loader2 className="h-10 w-10 text-emerald-600 animate-spin" aria-hidden="true" />
   486→                          <p className="text-sm font-medium text-slate-700 dark:text-slate-200">
   487→                            Removing background...
   488→                          </p>
   489→                          <p className="text-xs text-slate-500 dark:text-slate-400">
   490→                            Preserving full subject with clean edges
   491→                          </p>
   492→                        </div>
   493→                      </div>
   494→                    </div>
   495→                  )}
   496→
   497→                  {/* Progress bar */}
   498→                  <div className="space-y-2">
   499→                    <div className="flex items-center justify-between text-sm">
   500→                      <span className="text-slate-600 dark:text-slate-400">
   501→                        Progress
   502→                      </span>
   503→                      <span className="font-medium text-emerald-600 tabular-nums" aria-live="polite">
   504→                        {Math.round(progress)}%
   505→                      </span>
   506→                    </div>
   507→                    <Progress
   508→                      value={progress}
   509→                      className="h-2.5 bg-slate-100 dark:bg-slate-800 [&>[data-slot=progress-indicator]]:bg-emerald-600"
   510→                      aria-label={`Processing progress: ${Math.round(progress)}%`}
   511→                    />
   512→                  </div>
   513→
   514→                  <div className="flex justify-center">
   515→                    <Button
   516→                      variant="outline"
   517→                      onClick={handleReset}
   518→                      className="gap-2"
   519→                    >
   520→                      <RefreshCw className="h-4 w-4" aria-hidden="true" />
   521→                      Cancel
   522→                    </Button>
   523→                  </div>
   524→                </CardContent>
   525→              </Card>
   526→            </motion.section>
   527→          )}
   528→
   529→          {/* DONE STATE - Before/After */}
   530→          {state === 'done' && originalImage && processedImage && (
   531→            <motion.section
   532→              key="done"
   533→              aria-label="Background removal result"
   534→              initial={{ opacity: 0, y: 20 }}
   535→              animate={{ opacity: 1, y: 0 }}
   536→              exit={{ opacity: 0, y: -20 }}
   537→              transition={{ duration: 0.3 }}
   538→              className="mx-auto max-w-5xl"
   539→            >
   540→              {/* Success banner */}
   541→              <motion.div
   542→                initial={{ opacity: 0, scale: 0.95 }}
   543→                animate={{ opacity: 1, scale: 1 }}
   544→                transition={{ delay: 0.1 }}
   545→                className="mb-6 flex items-center gap-3 rounded-xl bg-emerald-50 dark:bg-emerald-950/30 border border-emerald-200 dark:border-emerald-800 p-4"
