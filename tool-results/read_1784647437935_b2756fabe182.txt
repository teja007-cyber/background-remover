    87→function formatFileSize(bytes: number) {
    88→  if (bytes < 1024) return `${bytes} B`
    89→  if (bytes < 1024 * 1024) return `${(bytes / 1024).toFixed(1)} KB`
    90→  return `${(bytes / (1024 * 1024)).toFixed(1)} MB`
    91→}

    93→function generateId() {
    94→  return Math.random().toString(36).slice(2, 10)

    97→function readFileAsDataUrl(file: File): Promise<string> {
    98→  return new Promise((resolve, reject) => {
    99→    const reader = new FileReader()
   100→    reader.onload = (e) => resolve(e.target?.result as string)
   101→    reader.onerror = reject
   102→    reader.readAsDataURL(file)
   103→  })
   104→}

   326→      }
   327→      return combined
   328→    })
   329→
   330→    if (newItems.length > 0) {
   331→      toast.success(`${newItems.length} image${newItems.length > 1 ? 's' : ''} added`)

   362→
   363→      setBatchItems(prev => prev.map(bi =>

   374→          method: 'POST',
   375→          body: formData,
   376→          signal: batchAbortRef.current?.signal,

   381→
   382→        if (!response.ok || !data.success) {
   383→          setBatchItems(prev => prev.map(bi =>
   384→            bi.id === item.id ? { ...bi, status: 'error' as const, error: data.error || 'Failed' } : bi
   385→          ))
   386→          continue
   387→        }
   388→
   389→        setBatchItems(prev => prev.map(bi =>
   390→          bi.id === item.id ? {
   391→            ...bi, status: 'done' as const,
   392→            processedDataUrl: data.image, processingTime: Number(elapsed),
   393→          } : bi
   394→        ))
   395→      } catch (err) {
   396→        if (err instanceof DOMException && err.name === 'AbortError') break
   397→        setBatchItems(prev => prev.map(bi =>
   398→          bi.id === item.id ? { ...bi, status: 'error' as const, error: 'Network error' } : bi
   399→        ))
   400→      }
   401→    }
   402→
   403→    setBatchProcessing(false)
   404→    setBatchCurrentIndex(-1)

   482→              Free &amp; Private
